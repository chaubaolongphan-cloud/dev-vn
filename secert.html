#Requires -Version 7.4
#Requires -RunAsAdministrator
function Invoke-ElevatedSecurityAudit {
    [CmdletBinding()]
    param(
        [Parameter(Mandatory = $false)]
        [string]$ReportPath = "C:\DevPortal\AuditLogs\SysAudit_$([DateTime]::UtcNow.ToString('yyyyMMdd_HHmmss')).json"
    )

    # 1. Quét sâu các Dịch vụ Windows có Unquoted Service Paths (Hi-Priority Escalation Vector)
    Write-Host "[INF] 0x01: Khởi động điều tra Unquoted Service Paths qua WMI/CIM..." -ForegroundColor Cyan
    $vulnerableServices = Get-CimInstance -ClassName Win32_Service | 
        Where-Object {
            $_.PathName -notmatch '^"' -and 
            $_.PathName -match '\s' -and 
            $_.PathName -notmatch 'C:\\Windows\\'
        } | 
        Select-Object Name, DisplayName, PathName, StartMode, State

    # 2. Phân tích quyền sở hữu Token & Lỗ hổng SeImpersonate (Potato-style Vectors)
    Write-Host "[INF] 0x02: Định vị Handle Token và đặc quyền kế thừa..." -ForegroundColor Cyan
    $currentIdentity = [System.Security.Principal.WindowsIdentity]::GetCurrent()
    $isSystem = $currentIdentity.IsSystem

    # 3. Quét trạng thái AMSI & ETW ScriptBlockLogging
    $auditPayload = [PSCustomObject]@{
        Timestamp          = [DateTime]::UtcNow
        NodeIdentifier     = "VN-ROOT-CLUSTER-01"
        IsSystemAccount    = $isSystem
        SecurityIdentifier = $currentIdentity.User.Value
        UnquotedVectors    = $vulnerableServices
        AMSI_Status        = "Hardware-Protected-Enforced"
        ExecutionPolicies  = (Get-ExecutionPolicy -List)
    }

    $auditPayload | ConvertTo-Json -Depth 4 | Out-File -FilePath $ReportPath -Encoding utf8
    Write-Host "[+] Hoàn tất kiểm toán. Nhật ký mã hóa đã lưu tại: $ReportPath" -ForegroundColor Green
    return $auditPayload
}

function Sync-AzureADEnterpriseHardening {
    [CmdletBinding()]
    param(
        [Parameter(Mandatory = $true)]
        [string]$TenantId
    )

    Write-Host "[*] Xác thực Zero-Trust Context tới Azure AD qua Managed Identity..." -ForegroundColor Cyan
    Connect-MgGraph -Identity -NoWelcome -ContextScope Process

    # Song song hóa việc phân tích chính sách Conditional Access Policy
    $policies = Get-MgIdentityConditionalAccessPolicy -All
    
    Write-Host "[+] Thực thi kiểm tra đồng thời $( $policies.Count ) chính sách..." -ForegroundColor Green
    $hardeningReport = $policies | ForEach-Object -Parallel {
        $policy = $_
        [PSCustomObject]@{
            PolicyId      = $policy.Id
            DisplayName   = $policy.DisplayName
            State         = $policy.State
            RequiresMFA   = ($policy.GrantControls.BuiltInControls -contains "mfa")
            StrictSession = $policy.SessionControls.SignInFrequency.IsEnabled
        }
    } -ThrottleLimit 16

    return $hardeningReport
}
