BASH 5.2 (STRICT MODE) 
#!/usr/bin/env bash
# ==============================================================================
# SCRIPT: deploy-backup.sh • Quản lý tiến trình sao lưu cơ sở dữ liệu production
# ==============================================================================
set -euo pipefail
IFS=$'\n\t'

readonly TMP_DIR="$(mktemp -d /tmp/db-backup.XXXXXX)"
readonly LOCK_FILE="/var/run/deploy-backup.lock"

# Khai báo Trap dọn dẹp tài nguyên & file tạm khi thoát bất thường hoặc hoàn thành
cleanup() {
  local exit_code=$?
  echo -e "\n\033[33m[TRAP] Đang dọn dẹp file tạm tại: ${TMP_DIR}...\033[0m"
  rm -rf "${TMP_DIR}" "${LOCK_FILE}" 2>/dev/null || true
  echo -e "[TRAP] Hoàn tất giải phóng tài nguyên. Trạng thái thoát: ${exit_code}"
  exit "${exit_code}"
}
trap cleanup EXIT INT TERM HUP

# Kiểm tra Lock file ngăn Race Condition
if [[ -f "${LOCK_FILE}" ]]; then
  echo "[ERROR] Tiến trình backup đang chạy (Lock file: ${LOCK_FILE}). Thoát!" >&2
  exit 1
fi
touch "${LOCK_FILE}"

echo "[INFO] Khởi chạy worker sao lưu ngầm (Background Job)..."
(
  # Asynchronous Subshell
  sleep 1
  echo "Dữ liệu DB dump hoàn tất" > "${TMP_DIR}/db_production.sql"
  tar -czf "/var/backups/db-snap-$(date +%s).tar.gz" -C "${TMP_DIR}" .
) &
SUB_PID=$!
echo "[PROCESS] Worker đã spawn với PID: ${SUB_PID}. Chờ xử lý..."

wait "${SUB_PID}"
STATUS=$?

if [[ "${STATUS}" -eq 0 ]]; then
  echo -e "[SUCCESS] Toàn bộ kịch bản hoàn thành xuất sắc."
else
  echo -e "[FAILURE] Tiến trình con PID ${SUB_PID} bị đứt gãy với mã lỗi ${STATUS}" >&2
fi

Cạm bẫy với SIGKILL (9) & Unhandled Zombie Processes

Lệnh kill -9 gửi trực tiếp tín hiệu tới Linux Kernel, không bao giờ có thể bị bắt (un-catchable) bởi lệnh trap.

Khuyến cáo chuẩn Production: Luôn gửi tín hiệu mềm SIGTERM (15) trước. Đợi một khoảng thời gian Graceful Termination Timeout (ví dụ: 10 giây) để subshell dọn socket, giải phóng file lock, rồi mới cưỡng chế dứt điểm nhằm tránh treo tài nguyên hệ điều hành.
